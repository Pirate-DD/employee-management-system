```java
package com.company.ems.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.List;

@Component
public class JwtTokenProvider {
    
    @Value("${jwt.secret}")
    private String secret;
    
    @Value("${jwt.access-token.expiration}")
    private long accessTokenExpiration;
    
    @Value("${jwt.refresh-token.expiration}")
    private long refreshTokenExpiration;
    
    private SecretKey getSigningKey() {
        byte[] keyBytes = secret.getBytes(StandardCharsets.UTF_8);
        return Keys.hmacShaKeyFor(keyBytes);
    }
    
    public String generateAccessToken(Long userId, String username, List<String> roles) {
        return Jwts.builder()
            .setSubject(username)
            .claim("userId", userId)
            .claim("roles", roles)
            .setIssuedAt(new Date())
            .setExpiration(new Date(System.currentTimeMillis() + accessTokenExpiration * 1000))
            .signWith(getSigningKey(), SignatureAlgorithm.HS256)
            .compact();
    }
    
    public String generateRefreshToken(Long userId) {
        return Jwts.builder()
            .setSubject("refresh")
            .claim("userId", userId)
            .setIssuedAt(new Date())
            .setExpiration(new Date(System.currentTimeMillis() + refreshTokenExpiration * 1000))
            .signWith(getSigningKey(), SignatureAlgorithm.HS256)
            .compact();
    }
    
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder()
                .setSigningKey(getSigningKey())
                .build()
                .parseClaimsJws(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            return false;
        }
    }
    
    public Long getUserIdFromToken(String token) {
        Claims claims = Jwts.parserBuilder()
            .setSigningKey(getSigningKey())
            .build()
            .parseClaimsJws(token)
            .getBody();
        return claims.get("userId", Long.class);
    }
    
    public String getUsernameFromToken(String token) {
        return Jwts.parserBuilder()
            .setSigningKey(getSigningKey())
            .build()
            .parseClaimsJws(token)
            .getBody()
            .getSubject();
    }
}
```

```java
package com.company.ems.exception;

import com.company.ems.common.Result;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {
    
    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<Result<Void>> handleBusinessException(BusinessException e) {
        log.warn("业务异常: {}", e.getMessage());
        return ResponseEntity.ok(Result.error(e.getCode(), e.getMessage()));
    }
    
    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<Result<Void>> handleResourceNotFoundException(ResourceNotFoundException e) {
        log.warn("资源不存在: {}", e.getMessage());
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
            .body(Result.error(404, e.getMessage()));
    }
    
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Result<Void>> handleValidationException(MethodArgumentNotValidException e) {
        String message = e.getBindingResult().getFieldErrors().stream()
            .map(error -> error.getField() + ": " + error.getDefaultMessage())
            .collect(Collectors.joining("; "));
        log.warn("参数校验失败: {}", message);
        return ResponseEntity.badRequest()
            .body(Result.error(1001, message));
    }
    
    @ExceptionHandler(Exception.class)
    public ResponseEntity<Result<Void>> handleException(Exception e) {
        log.error("系统异常", e);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(Result.error(5001, "系统内部错误，请联系管理员"));
    }
}
```

```java
package com.company.ems.aspect;

import com.company.ems.annotation.OperateLog;
import com.company.ems.entity.OperateLog;
import com.company.ems.service.OperateLogService;
import com.company.ems.util.SecurityUtils;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.Around;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;

@Slf4j
@Aspect
@Component
public class OperateLogAspect {
    
    @Autowired
    private OperateLogService logService;
    
    @Autowired
    private ObjectMapper objectMapper;
    
    @Around("@annotation(operateLog)")
    public Object around(ProceedingJoinPoint joinPoint, OperateLog operateLog) throws Throwable {
        long startTime = System.currentTimeMillis();
        String userName = SecurityUtils.getCurrentUsername();
        String ip = SecurityUtils.getCurrentIp();
        
        try {
            Object result = joinPoint.proceed();
            saveLog(operateLog, joinPoint, userName, ip, startTime, result, null);
            return result;
        } catch (Exception e) {
            saveLog(operateLog, joinPoint, userName, ip, startTime, null, e);
            throw e;
        }
    }
    
    @Async
    protected void saveLog(OperateLog operateLog, ProceedingJoinPoint joinPoint,
                           String userName, String ip, long startTime, Object result, Exception e) {
        try {
            long duration = System.currentTimeMillis() - startTime;
            
            OperateLog logEntity = OperateLog.builder()
                .userName(userName)
                .module(operateLog.module())
                .action(operateLog.action())
                .description(operateLog.description())
                .requestMethod(joinPoint.getSignature().getName())
                .requestParams(toJson(joinPoint.getArgs()))
                .responseResult(toJson(result))
                .ip(ip)
                .duration(duration)
                .status(e == null ? 1 : 0)
                .errorMsg(e != null ? e.getMessage() : null)
                .createTime(LocalDateTime.now())
                .build();
            
            logService.save(logEntity);
        } catch (Exception ex) {
            log.error("保存操作日志失败", ex);
        }
    }
    
    private String toJson(Object obj) {
        if (obj == null) return null;
        try {
            return objectMapper.writeValueAsString(obj);
        } catch (JsonProcessingException e) {
            return obj.toString();
        }
    }
}
```

```java
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class EmployeeServiceImpl implements EmployeeService {
    
    private final EmployeeRepository employeeRepository;
    private final UserRepository userRepository;
    private final UserRoleService userRoleService;
    private final PasswordEncoder passwordEncoder;
    
    @Override
    @Transactional
    public EmployeeVO createEmployee(EmployeeCreateDTO dto) {
        // 检查工号唯一性
        if (employeeRepository.existsByEmpNo(dto.getEmpNo())) {
            throw new BusinessException(1002, "工号已存在");
        }
        
        // 创建用户账号
        User user = User.builder()
            .username(dto.getUsername())
            .password(passwordEncoder.encode(dto.getPassword()))
            .realName(dto.getName())
            .email(dto.getEmail())
            .phone(dto.getPhone())
            .status(1)
            .build();
        User savedUser = userRepository.save(user);
        
        // 创建员工信息
        Employee employee = Employee.builder()
            .empNo(dto.getEmpNo())
            .userId(savedUser.getId())
            .deptId(dto.getDeptId())
            .name(dto.getName())
            .gender(dto.getGender())
            .birthDate(dto.getBirthDate())
            .idCard(dto.getIdCard())
            .phone(dto.getPhone())
            .email(dto.getEmail())
            .avatar(dto.getAvatar())
            .education(dto.getEducation())
            .politicalStatus(dto.getPoliticalStatus())
            .nativePlace(dto.getNativePlace())
            .address(dto.getAddress())
            .emergencyContact(dto.getEmergencyContact())
            .emergencyPhone(dto.getEmergencyPhone())
            .entryDate(dto.getEntryDate())
            .position(dto.getPosition())
            .level(dto.getLevel())
            .employeeStatus(2) // 试用期
            .build();
        Employee savedEmployee = employeeRepository.save(employee);
        
        // 分配基础角色
        userRoleService.assignRole(savedUser.getId(), "EMPLOYEE");
        
        return EmployeeVO.convert(savedEmployee);
    }
    
    @Override
    public EmployeeVO getEmployeeById(Long id) {
        Employee employee = employeeRepository.findById(id)
            .orElseThrow(() -> new ResourceNotFoundException("员工不存在"));
        return EmployeeVO.convert(employee);
    }
    
    @Override
    public Page<EmployeeListVO> getEmployeeList(EmployeeQueryDTO query, Pageable pageable) {
        return employeeRepository.findAll(query.toSpecification(), pageable)
            .map(EmployeeListVO::convert);
    }
    
    @Override
    @Transactional
    public EmployeeVO updateEmployee(Long id, EmployeeUpdateDTO dto) {
        Employee employee = employeeRepository.findById(id)
            .orElseThrow(() -> new ResourceNotFoundException("员工不存在"));
        
        // 更新非空字段
        if (dto.getName() != null) employee.setName(dto.getName());
        if (dto.getGender() != null) employee.setGender(dto.getGender());
        if (dto.getPhone() != null) employee.setPhone(dto.getPhone());
        if (dto.getEmail() != null) employee.setEmail(dto.getEmail());
        if (dto.getDeptId() != null) employee.setDeptId(dto.getDeptId());
        if (dto.getPosition() != null) employee.setPosition(dto.getPosition());
        if (dto.getLevel() != null) employee.setLevel(dto.getLevel());

                user.setStatus(0);
                userRepository.save(user);
            });
        }
    }
```

```java
@Service
@RequiredArgsConstructor
public class AttendanceServiceImpl implements AttendanceService {
    
    private final AttendanceRepository attendanceRepository;
    private final EmployeeRepository employeeRepository;
    private final LeaveRepository leaveRepository;
    
    @Value("${attendance.config.work-start:09:00}")
    private String workStart;
    
    @Value("${attendance.config.work-end:18:00}")
    private String workEnd;
    
    @Value("${attendance.config.late-threshold:5}")
    private int lateThreshold;
    
    @Override
    @Transactional
    public AttendanceVO checkIn(Long empId, CheckInDTO dto) {
        Employee employee = employeeRepository.findById(empId)
            .orElseThrow(() -> new ResourceNotFoundException("员工不存在"));
        
        LocalDate today = LocalDate.now();
        LocalDateTime now = LocalDateTime.now();
        
        // 获取或创建今日考勤记录
        Attendance attendance = attendanceRepository
            .findByEmpIdAndAttendanceDate(empId, today)
            .orElseGet(() -> {
                Attendance newAtt = new Attendance();
                newAtt.setEmpId(empId);
                newAtt.setAttendanceDate(today);
                return newAtt;
            });
        
        if ("IN".equals(dto.getCheckType())) {
            // 上班打卡
            if (attendance.getCheckInTime() != null) {
                throw new BusinessException(3001, "今日已打过上班卡");
            }
            
            attendance.setCheckInTime(now);
            
            // 判断迟到
            LocalTime workStartTime = LocalTime.parse(workStart);
            long lateMinutes = Duration.between(workStartTime, now.toLocalTime()).toMinutes();
            
            if (now.toLocalTime().isAfter(workStartTime)) {
                if (lateMinutes > lateThreshold) {
                    attendance.setIsLate(1);
                    attendance.setLateMinutes((int) lateMinutes);
                    attendance.setWorkStatus(2); // 迟到
                } else {
                    attendance.setWorkStatus(1); // 正常
                }
            } else {
                attendance.setWorkStatus(1); // 正常
            }
        } else if ("OUT".equals(dto.getCheckType())) {
            // 下班打卡
            if (attendance.getCheckOutTime() != null) {
                throw new BusinessException(3001, "今日已打过下班卡");
            }
            
            if (attendance.getCheckInTime() == null) {
                throw new BusinessException(3001, "请先打上班卡");
            }
            
            attendance.setCheckOutTime(now);
            
            // 计算工作时长
            double hours = Duration.between(attendance.getCheckInTime(), now).toMinutes() / 60.0;
            attendance.setWorkHours(Math.round(hours * 100.0) / 100.0);
            
            // 判断早退
            LocalTime workEndTime = LocalTime.parse(workEnd);
            if (now.toLocalTime().isBefore(workEndTime)) {
                long earlyMinutes = Duration.between(now.toLocalTime(), workEndTime).toMinutes();
                if (earlyMinutes > lateThreshold) {
                    attendance.setIsEarly(1);
                    attendance.setEarlyMinutes((int) earlyMinutes);
                }
            }
        }
        
        Attendance saved = attendanceRepository.save(attendance);
        return AttendanceVO.convert(saved);
    }
    
    @Override
    public MonthlyStatsVO getMonthlyStats(Long empId, String month) {
        LocalDate monthStart = LocalDate.parse(month + "-01");
        LocalDate monthEnd = monthStart.plusMonths(1).minusDays(1);
        
        List<Attendance> attendances = attendanceRepository
            .findByEmpIdAndAttendanceDateBetween(empId, monthStart, monthEnd);
        
        MonthlyStatsVO stats = MonthlyStatsVO.builder()
            .month(month)
            .totalDays(monthStart.lengthOfMonth())
            .normalDays(0)
            .lateDays(0)
            .leaveDays(0)
            .overtimeHours(0.0)
            .build();
        
        for (Attendance att : attendances) {
            switch (att.getWorkStatus()) {
                case 1 -> stats.setNormalDays(stats.getNormalDays() + 1);
                case 2 -> stats.setLateDays(stats.getLateDays() + 1);
                case 4 -> stats.setLeaveDays(stats.getLeaveDays() + 1);
                case 5 -> stats.setOvertimeHours(stats.getOvertimeHours() + att.getWorkHours());
            }
        }
        
        return stats;
    }
}
```

```java
                                            boolean isHoliday) {
        BigDecimal baseSalary = plan.getBaseSalary();
        BigDecimal performanceSalary = plan.getPerformanceSalary();
        BigDecimal dailySalary = baseSalary.add(performanceSalary)
            .divide(workDaysPerMonth, 2, RoundingMode.HALF_UP);
        
        // 1. 应发工资组成
        BigDecimal grossSalary = baseSalary
            .add(performanceSalary)
            .add(plan.getPostAllowance())
            .add(plan.getMealAllowance())
            .add(plan.getCommunicationAllowance())
            .add(plan.getOtherAllowance());
        
        // 2. 加班工资
        BigDecimal overtimePay = BigDecimal.ZERO;
        if (overtimeHours > 0) {
            BigDecimal rate = isHoliday ? overtimeRateHoliday : 
                            isWeekend ? overtimeRateWeekend : overtimeRateWorkday;
            BigDecimal hourlyRate = baseSalary
                .divide(workDaysPerMonth, 2, RoundingMode.HALF_UP)
                .divide(BigDecimal.valueOf(8), 2, RoundingMode.HALF_UP);
            overtimePay = hourlyRate.multiply(BigDecimal.valueOf(overtimeHours))
                .multiply(rate).setScale(2, RoundingMode.HALF_UP);
            grossSalary = grossSalary.add(overtimePay);
        }
        
        // 3. 请假扣款
        BigDecimal leaveDeduction = dailySalary.multiply(BigDecimal.valueOf(leaveDays))
            .setScale(2, RoundingMode.HALF_UP);
        BigDecimal lateEarlyDeduction = BigDecimal.ZERO;
        
        // 4. 个人社保公积金
        BigDecimal pensionDeduction = plan.getSocialInsuranceBase()
            .multiply(pensionRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal medicalDeduction = plan.getSocialInsuranceBase()
            .multiply(medicalRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal unemploymentDeduction = plan.getSocialInsuranceBase()
            .multiply(unemploymentRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal housingFundDeduction = plan.getHousingFundBase()
            .multiply(housingFundRate).setScale(2, RoundingMode.HALF_UP);
        
        // 5. 应纳税所得额
        BigDecimal taxableIncome = grossSalary
            .subtract(leaveDeduction)
            .subtract(taxThreshold)
            .subtract(pensionDeduction)
            .subtract(medicalDeduction)
            .subtract(unemploymentDeduction)
            .subtract(housingFundDeduction);
        
        // 6. 个税计算
        BigDecimal taxDeduction = calculateTax(taxableIncome);
        
        // 7. 实发工资
        BigDecimal totalDeduction = leaveDeduction
            .add(lateEarlyDeduction)
            .add(pensionDeduction)
            .add(medicalDeduction)
            .add(unemploymentDeduction)
            .add(housingFundDeduction)
            .add(taxDeduction);
        BigDecimal netSalary = grossSalary.subtract(totalDeduction)
            .setScale(2, RoundingMode.HALF_UP);
        
        return SalaryCalculateResult.builder()
            .baseSalary(baseSalary)
            .performanceSalary(performanceSalary)
            .postAllowance(plan.getPostAllowance())
            .mealAllowance(plan.getMealAllowance())
            .communicationAllowance(plan.getCommunicationAllowance())
            .otherAllowance(plan.getOtherAllowance())
            .overtimePay(overtimePay)
            .leaveDeduction(leaveDeduction)
            .lateEarlyDeduction(lateEarlyDeduction)
            .grossSalary(grossSalary.setScale(2, RoundingMode.HALF_UP))
            .pensionDeduction(pensionDeduction)
            .medicalDeduction(medicalDeduction)
            .unemploymentDeduction(unemploymentDeduction)
            .housingFundDeduction(housingFundDeduction)
            .taxDeduction(taxDeduction)
            .totalDeduction(totalDeduction.setScale(2, RoundingMode.HALF_UP))
            .netSalary(netSalary)
            .build();
    }
    
    private BigDecimal calculateTax(BigDecimal taxableIncome) {
        if (taxableIncome.compareTo(BigDecimal.ZERO) <= 0) {
            return BigDecimal.ZERO;
        }
        
        BigDecimal tax;
        if (taxableIncome.compareTo(new BigDecimal("3000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.03"));
        } else if (taxableIncome.compareTo(new BigDecimal("12000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.10")).subtract(new BigDecimal("210"));
        } else if (taxableIncome.compareTo(new BigDecimal("25000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.20")).subtract(new BigDecimal("1410"));
        } else if (taxableIncome.compareTo(new BigDecimal("35000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.25")).subtract(new BigDecimal("2660"));
        } else if (taxableIncome.compareTo(new BigDecimal("55000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.30")).subtract(new BigDecimal("4410"));
        } else if (taxableIncome.compareTo(new BigDecimal("80000")) <= 0) {
            tax = taxableIncome.multiply(new BigDecimal("0.35")).subtract(new BigDecimal("7160"));
        } else {
            tax = taxableIncome.multiply(new BigDecimal("0.45")).subtract(new BigDecimal("15160"));
        }
        
        return tax.setScale(2, RoundingMode.HALF_UP);
    }
}
```
